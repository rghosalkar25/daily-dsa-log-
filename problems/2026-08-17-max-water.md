Approach-Two Pointers: Start with pointers at both ends of the array. Calculate the area between them and move the pointer with the smaller height inward. Keep track of the maximum area found.

Time: O(n)
Space: O(1)

<!-- Python -->
def maxArea(height):
    left = 0
    right = len(height) - 1
    max_area = 0

    while left < right:
        width = right - left
        area = min(height[left], height[right]) * width
        max_area = max(max_area, area)

        if height[left] < height[right]:
            left += 1
        else:
            right -= 1

    return max_area
