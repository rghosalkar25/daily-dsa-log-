Approach-Greedy: Traverse the prices and add every positive price difference between consecutive days. This captures all profitable transactions without needing to track individual buy and sell days.

Time: O(n)
Space: O(1)

<!-- Python -->
def maxProfit(prices):
    profit = 0

    for i in range(1, len(prices)):
        if prices[i] > prices[i - 1]:
            profit += prices[i] - prices[i - 1]

    return profit
